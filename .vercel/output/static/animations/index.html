/**
 * ============================================================================
 * ANIME.JS v4 VAULT — 01. CARGA SENCILLA (MORPHING BAR & PULSE)
 * Archivo: animations/loaders/simple-pulse.js
 * Documentación: https://animejs.com
 * ============================================================================
 * Uso:
 *   import { createSimplePulseLoader } from './animations/loaders/simple-pulse.js';
 *   const ctrl = createSimplePulseLoader('#mi-loader');
 */
import { animate } from 'animejs';

export function createSimplePulseLoader(containerSelector) {
  const root = typeof containerSelector === 'string'
    ? document.querySelector(containerSelector)
    : containerSelector;
  if (!root) return null;

  const bar = root.querySelector('.cxd-loader-bar-fill') || root;

  return animate(bar, {
    scaleX: [0.12, 0.85, 0.25, 1],
    translateX: ['-35%', '15%', '-10%', '0%'],
    opacity: [0.65, 1, 0.8, 1],
    duration: 1600,
    ease: 'inOutQuad',
    loop: true,
    alternate: true

 * ============================================================================
 * ANIME.JS v4 VAULT — 05. STEP-BY-STEP WIZARD TIMELINE
 * Archivo: animations/workflows/step-by-step.js
 * ============================================================================
 * Orquesta flujos paso a paso (Onboarding, Checkout, Deploy Pipeline):
 *  - Avanza la barra de progreso con física suave
 *  - Ilumina y escala el nodo del paso activo
 *  - Anima la entrada del panel de contenido correspondiente
 */
import { animate, createTimeline } from 'animejs';

export function animateToStep(container, stepIndex, totalSteps = 4) {
  const root = typeof container === 'string' ? document.querySelector(container) : container;
  if (!root) return null;

  const progressFill = root.querySelector('[data-step-progress]');
  const nodes = Array.from(root.querySelectorAll('[data-step-node]'));
  const panels = Array.from(root.querySelectorAll('[data-step-panel]'));

  const percentage = Math.min(100, Math.max(0, (stepIndex / (totalSteps - 1)) * 100));

  const tl = createTimeline({
    defaults: { ease: 'outQuart', duration: 480 }
  });

  if (progressFill) {
    tl.add(progressFill, {
      width: `${percentage}%`
    });
  }

  nodes.forEach((node, idx) => {
    const isCompleted = idx < stepIndex;
    const isCurrent = idx === stepIndex;
    node.classList.toggle('is-completed', isCompleted);
    node.classList.toggle('is-current', isCurrent);

    if (isCurrent) {
      animate(node, {
        scale: [0.82, 1.14, 1],
        duration: 520,
        ease: 'outBack'
      });
    }
  });

  panels.forEach((panel, idx) => {
    const active = idx === stepIndex;
    panel.style.display = active ? 'block' : 'none';
    if (active) {
      animate(panel, {
        opacity: [0, 1],
        y: ['12px', '0px'],
        duration: 420,
        ease: 'outCubic'
      });
    }
  });

  return tl;
}

/**
 * ============================================================================
 * ANIME.JS v4 VAULT — 09. CHRONO TIMER DE ALTA PRECISIÓN (`createTimer`)
 * Archivo: animations/timers/chrono-timer.js
 * ============================================================================
 * Usa la API nativa `createTimer()` de Anime.js v4 para gestionar un bucle de
 * tiempo exacto con control de play, pause, reset y sincronización de anillo SVG.
 */
import { createTimer, animate } from 'animejs';

export class ChronoTimerController {
  constructor({ displayEl, ringEl, durationMs = 15000, onComplete }) {
    this.displayEl = displayEl;
    this.ringEl = ringEl;
    this.durationMs = durationMs;
    this.circumference = 283; // 2 * PI * 45

    this.timer = createTimer({
      duration: durationMs,
      autoplay: false,
      onUpdate: (self) => {
        const elapsed = self.currentTime;
        const remaining = Math.max(0, durationMs - elapsed);
        const seconds = Math.floor(remaining / 1000);
        const centis = Math.floor((remaining % 1000) / 10);

        if (this.displayEl) {
          this.displayEl.textContent = `${String(seconds).padStart(2, '0')}.${String(centis).padStart(2, '0')}s`;
        }

        if (this.ringEl) {
          const progress = Math.min(1, elapsed / durationMs);
          this.ringEl.style.strokeDashoffset = String(this.circumference * progress);
        }
      },
      onComplete: () => {
        if (this.displayEl) {
          animate(this.displayEl, {
            scale: [1, 1.18, 1],
            duration: 450,
            ease: 'outBack'
          });
        }
        onComplete?.();
      }
    });
  }

  play() {
    this.timer.play();
  }

  pause() {
    this.timer.pause();
  }

  restart() {
    this.timer.restart();
  }
}
